function step(){
  const all=[cue,...balls];
  for(const b of all){
    if(!b.on)continue;
    b.x+=b.vx;b.y+=b.vy;
    b.vx*=.986;b.vy*=.986;
    if(Math.hypot(b.vx,b.vy)<.03){b.vx=0;b.vy=0}
    // cushions
    if(b.x<FELT.x+BR){b.x=FELT.x+BR;b.vx=-b.vx*.82;railSnd()}
    if(b.x>FELT.x+FELT.w-BR){b.x=FELT.x+FELT.w-BR;b.vx=-b.vx*.82;railSnd()}
    if(b.y<FELT.y+BR){b.y=FELT.y+BR;b.vy=-b.vy*.82;railSnd()}
    if(b.y>FELT.y+FELT.h-BR){b.y=FELT.y+FELT.h-BR;b.vy=-b.vy*.82;railSnd()}
  }
  // ball-ball
  for(let i=0;i<all.length;i++)for(let j=i+1;j<all.length;j++){
    const a=all[i],b=all[j];if(!a.on||!b.on)continue;
    const dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy);
    if(d>0&&d<BR*2){
      const nx=dx/d,ny=dy/d,ov=(BR*2-d)/2;
      a.x-=nx*ov;a.y-=ny*ov;b.x+=nx*ov;b.y+=ny*ov;
      const avn=a.vx*nx+a.vy*ny,bvn=b.vx*nx+b.vy*ny,dv=avn-bvn;
      if(dv>0){
        a.vx-=dv*nx;a.vy-=dv*ny;b.vx+=dv*nx;b.vy+=dv*ny;
        if((a===cue||b===cue)&&!shotFirst)shotFirst=a===cue?b.n:a.n;
        clickSnd(Math.min(1,dv/12));
      }
    }
  }
  // pockets
  for(const p of pockets())for(const b of all){
    if(!b.on)continue;
    if(Math.hypot(b.x-p.x,b.y-p.y)<PR*.92){
      b.on=false;b.vx=b.vy=0;pocketSnd();
      if(b===cue)shotCue=true;else shotPotted.push(b.n);
    }
  }
}

function drawBall(b){
  if(!b.on)return;
  const g=ctx.createRadialGradient(b.x-BR*.4,b.y-BR*.4,BR*.15,b.x,b.y,BR);
  if(b.n===0){g.addColorStop(0,'#fff');g.addColorStop(1,'#c8cfe8')}

  ctx.fillStyle=g;ctx.beginPath();ctx.arc(b.x,b.y,BR,0,7);ctx.fill();
