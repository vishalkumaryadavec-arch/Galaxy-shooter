function reset(full){
  if(full){myScore=0;aiScore=0;over=false}
  me={x:W/2,y:H*.82,r:28,vx:0,vy:0,tx:W/2,ty:H*.82};

function clampPuck(){const sp=Math.hypot(puck.vx,puck.vy),mx=16;if(sp>mx){puck.vx*=mx/sp;puck.vy*=mx/sp}}
function hitPad(p){
  const dx=puck.x-p.x,dy=puck.y-p.y,d=Math.hypot(dx,dy),rr=puck.r+p.r;
  if(d<rr&&d>0){
    const nx=dx/d,ny=dy/d;
    puck.x=p.x+nx*(rr+1);puck.y=p.y+ny*(rr+1);
    const rel=(puck.vx-p.vx)*nx+(puck.vy-p.vy)*ny;
    if(rel<0){puck.vx-=2*rel*nx;puck.vy-=2*rel*ny}
    puck.vx+=p.vx*.35;puck.vy+=p.vy*.35;
    clampPuck();beep(300,.06,'square',.05);if(navigator.vibrate)navigator.vibrate(20);
  }
}
function goalW(){return W*.42}
function update(){
  if(over)return;
  if(serveT>0){serveT--;return}
  // player paddle follows touch (bottom half only)
  const ox=me.x,oy=me.y;
