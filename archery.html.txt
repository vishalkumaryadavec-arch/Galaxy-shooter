<script>
const cv=document.getElementById('c'),ctx=cv.getContext('2d'),$=id=>document.getElementById(id);
let W,H;function resize(){W=cv.width=innerWidth;H=cv.height=innerHeight}resize();addEventListener('resize',resize);
let AC;function beep(f,d,t='square',v=.05){try{AC=AC||new(window.AudioContext||window.webkitAudioContext)();const o=AC.createOscillator(),g=AC.createGain();o.type=t;o.frequency.value=f;g.gain.value=v;o.connect(g);g.connect(AC.destination);o.start();g.gain.exponentialRampToValueAtTime(.0001,AC.currentTime+d);o.stop(AC.currentTime+d)}catch(e){}}
function whoosh(){try{AC=AC||new(window.AudioContext||window.webkitAudioContext)();const o=AC.createOscillator(),g=AC.createGain();o.type='sawtooth';o.frequency.setValueAtTime(900,AC.currentTime);o.frequency.exponentialRampToValueAtTime(150,AC.currentTime+.25);g.gain.value=.05;o.connect(g);g.connect(AC.destination);o.start();o.stop(AC.currentTime+.25)}catch(e){}}
let best=+localStorage.getItem('ar_best')||0;
const ARROWS=10;
let score,arrowsLeft,round,over,msg='',msgT=0;
let bow,arrow,target,wind,pull=null,result=null,resultT=0,particles=[];
let stars=Array.from({length:70},()=>({x:Math.random()*innerWidth,y:Math.random()*innerHeight,s:Math.random()*2+.3}));
function newTarget(){
  const R=Math.max(34,Math.min(52,W*.11));
  target={x:W-60-R,y:H*.4,r:R,vy:(1.2+round*.35)*(Math.random()<.5?1:-1),minY:H*.16,maxY:H*.66,hitFlash:0};
  wind=+( (Math.random()*4-2).toFixed(1) );
  $('wind').textContent='🌬️ Hawa: '+(Math.abs(wind)<.2?'shaant':wind>0?'niche '+Math.abs(wind):'upar '+Math.abs(wind));
}
function setup(){
  score=0;arrowsLeft=ARROWS;round=1;over=false;arrow=null;result=null;pull=null;msg='';particles=[];
  bow={x:56,y:H*.55,r:42};
  newTarget();
  $('sc').textContent='Score 0';$('ar').textContent='🏹 '+arrowsLeft;
  $('over').classList.add('hide');
}
function boom(x,y,col,n){for(let i=0;i<n;i++){const a=Math.random()*6.28,s=Math.random()*4+1;particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,l:26+Math.random()*14,col})}}
function shoot(dx,dy){
  const d=Math.hypot(dx,dy);if(d<18)return;
  const pw=Math.min(d/11,17);
  arrow={x:bow.x,y:bow.y,vx:-dx/d*pw*1.9,vy:-dy/d*pw*1.9}; // slingshot: arrow flies OPPOSITE the pull
  arrowsLeft--;$('ar').textContent='🏹 '+arrowsLeft;
  whoosh();if(navigator.vibrate)navigator.vibrate(20);
  pull=null;
}
function update(){
  // target move
  if(target.hitFlash>0)target.hitFlash--;
  target.y+=target.vy;
  if(target.y<target.minY){target.y=target.minY;target.vy=Math.abs(target.vy)}
  if(target.y>target.maxY){target.y=target.maxY;target.vy=-Math.abs(target.vy)}
  // arrow flight
  if(arrow){
    arrow.vy+=.16+wind*.05; // gravity + hawa
    arrow.vx*=.998;
    arrow.x+=arrow.vx;arrow.y+=arrow.vy;
    // hit target plane?
    if(!result&&arrow.x+18>=target.x-target.r&&arrow.x-18<=target.x+target.r&&Math.abs(arrow.x-target.x)<14){
      const dy=arrow.y-target.y,ad=Math.abs(dy);
      let pts=0;
      if(ad<=target.r*.22)pts=10;else if(ad<=target.r*.5)pts=7;else if(ad<=target.r*.78)pts=4;else if(ad<=target.r)pts=1;
      if(pts>0){
        result={pts,x:target.x-target.r*.5,y:arrow.y};resultT=90;
        score+=pts;$('sc').textContent='Score '+score;
        target.hitFlash=20;
        beep(pts>=10?1200:pts>=7?900:pts>=4?700:500,.15,pts>=10?'sine':'triangle',.07);
        if(pts>=10){boom(target.x,target.y,'#ff0',20);msg='BULLSEYE! 🎯';msgT=70}
        if(navigator.vibrate)navigator.vibrate(pts>=10?[60,40,60]:30);
      }
      arrow=null;nextArrow();
      return;
    }
    if(arrow.x>W+40||arrow.y>H+40||arrow.y<-40){arrow=null;result={pts:0,x:W*.6,y:Math.min(H*.8,Math.max(60,arrow?arrow.y:H/2))};resultT=70;nextArrow()}
  }
  if(resultT>0){resultT--;if(resultT===0)result=null}
  if(msgT>0)msgT--;
  particles.forEach(p=>{p.x+=p.vx;p.y+=p.vy;p.vx*=.95;p.vy*=.95;p.l--});particles=particles.filter(p=>p.l>0);
}
function nextArrow(){
  if(arrowsLeft<=0){setTimeout(gameOver,900);return}
  round=ARROWS-arrowsLeft+1;
  setTimeout(()=>{if(!over)newTarget()},600);
}
function gameOver(){
  over=true;
  best=Math.max(best,score);localStorage.setItem('ar_best',best);
  $('fs').textContent=score;$('fb').textContent=best;
  $('rank').textContent=score>=80?'🏆 DHANURDHARI! Legend shot!':score>=55?'🥇 Khiladi level aim!':score>=30?'🥈 Achhi practice ho rahi hai':'🎯 Practice makes perfect!';
  $('over').classList.remove('hide');
  beep(880,.2,'sine',.07);setTimeout(()=>beep(1100,.25,'sine',.07),200);
}
function drawBow(){
  const py=pull?pull.y-bow.y:0,px=pull?pull.x-bow.x:0;
  const pd=Math.min(Math.hypot(px,py),90);
  const ang=pull?Math.atan2(py,px):0;
  ctx.save();ctx.translate(bow.x,bow.y);
  // bow arc
  ctx.strokeStyle='#f0f';ctx.lineWidth=5;ctx.shadowBlur=16;ctx.shadowColor='#f0f';
  ctx.beginPath();ctx.arc(0,0,bow.r,-1.1,1.1);ctx.stroke();
  // string
  const tx=Math.cos(0)*bow.r,topX=Math.cos(-1.1)*bow.r,topY=Math.sin(-1.1)*bow.r,botX=Math.cos(1.1)*bow.r,botY=Math.sin(1.1)*bow.r;
  ctx.strokeStyle='#dfe6ff';ctx.lineWidth=2;ctx.shadowBlur=6;ctx.shadowColor='#fff';
  const spx=pull?px/Math.max(pd,1)*pd*.9:0,spy=pull?py/Math.max(pd,1)*pd*.9:0;
  ctx.beginPath();ctx.moveTo(topX,topY);ctx.lineTo(spx,spy);ctx.lineTo(botX,botY);ctx.stroke();
  // nocked arrow
  if(!arrow&&arrowsLeft>0&&!over){
    const aimA=pull?Math.atan2(-spy,-spx):0;
    ctx.save();ctx.rotate(aimA);
    ctx.shadowBlur=10;ctx.shadowColor='#0ff';ctx.strokeStyle='#0ff';ctx.lineWidth=3;
    const al=52+ (pull?pd*.35:0);
    ctx.beginPath();ctx.moveTo(pull?-pd*.4:6,0);ctx.lineTo(pull?-pd*.4+al:6+al,0);ctx.stroke();
    ctx.fillStyle='#0ff';ctx.beginPath();ctx.moveTo(pull?-pd*.4+al+12:6+al+12,0);ctx.lineTo(pull?-pd*.4+al:6+al,-5);ctx.lineTo(pull?-pd*.4+al:6+al,5);ctx.closePath();ctx.fill();
    ctx.restore();
    // trajectory preview while pulling
    if(pull&&pd>15){
      const d=Math.hypot(px,py),pw=Math.min(d/11,17);
      let vx=-px/d*pw*1.9,vy=-py/d*pw*1.9,xx=0,yy=0;
      ctx.fillStyle='rgba(0,255,255,.5)';ctx.shadowBlur=0;
      for(let i=0;i<14;i++){vy+=.16+wind*.05;vx*=.998;xx+=vx;yy+=vy;
        if(i%2===0){ctx.beginPath();ctx.arc(xx,yy,2.2,0,6.29);ctx.fill()}}
    }
  }
  ctx.restore();
}
function draw(){
  ctx.fillStyle='#050816';ctx.fillRect(0,0,W,H);
  ctx.fillStyle='#fff';stars.forEach(s=>{ctx.globalAlpha=s.s/2.3;ctx.fillRect(s.x,s.y,s.s,s.s*2)});ctx.globalAlpha=1;
  // ground line
  ctx.strokeStyle='rgba(0,255,255,.2)';ctx.beginPath();ctx.moveTo(0,H*.88);ctx.lineTo(W,H*.88);ctx.stroke();
  // target stand
  ctx.strokeStyle='#8a93b8';ctx.lineWidth=5;
  ctx.beginPath();ctx.moveTo(target.x,target.y+target.r);ctx.lineTo(target.x,H*.88);ctx.stroke();
  // target rings
  const rings=[[1,'#e8ecff'],[.78,'#20242e'],[.5,'#ff2d55'],[.22,'#ffd60a']];
  ctx.save();if(target.hitFlash>0){ctx.shadowBlur=26;ctx.shadowColor='#ff0'}
  for(const[k,c]of rings){ctx.fillStyle=c;ctx.beginPath();ctx.arc(target.x,target.y,target.r*k,0,6.29);ctx.fill();
    ctx.strokeStyle='rgba(255,255,255,.25)';ctx.lineWidth=1;ctx.stroke()}
  ctx.restore();
  drawBow();
  // flying arrow
  if(arrow){const a=Math.atan2(arrow.vy,arrow.vx);
    ctx.save();ctx.translate(arrow.x,arrow.y);ctx.rotate(a);
    ctx.shadowBlur=10;ctx.shadowColor='#0ff';ctx.strokeStyle='#0ff';ctx.lineWidth=3;
    ctx.beginPath();ctx.moveTo(-18,0);ctx.lineTo(14,0);ctx.stroke();
    ctx.fillStyle='#0ff';ctx.beginPath();ctx.moveTo(20,0);ctx.lineTo(10,-4);ctx.lineTo(10,4);ctx.closePath();ctx.fill();
    ctx.fillStyle='#f0f';ctx.fillRect(-20,-4,6,8);
    ctx.restore()}
  // result popup
  if(result){ctx.font='bold 26px Arial';ctx.textAlign='center';
    ctx.fillStyle=result.pts>=10?'#ff0':result.pts>0?'#0ff':'#f66';
    ctx.shadowBlur=14;ctx.shadowColor=ctx.fillStyle;
    ctx.fillText(result.pts>0?'+'+result.pts:'MISS 😅',result.x,result.y-40);ctx.shadowBlur=0}
  if(msgT>0){ctx.font='bold 30px Arial';ctx.textAlign='center';ctx.fillStyle='#ff0';ctx.shadowBlur=18;ctx.shadowColor='#ff0';
    ctx.globalAlpha=Math.min(1,msgT/25);ctx.fillText(msg,W/2,H*.3);ctx.globalAlpha=1;ctx.shadowBlur=0}
  particles.forEach(p=>{ctx.globalAlpha=p.l/40;ctx.fillStyle=p.col;ctx.fillRect(p.x,p.y,3,3)});ctx.globalAlpha=1;
  // hint
  if(!pull&&!arrow&&arrowsLeft===ARROWS){ctx.font='14px Arial';ctx.textAlign='center';ctx.fillStyle='#9ad';
    ctx.fillText('Bow ke paas dabao, PICHE khincho, chhodo = shot 🏹',W/2,H*.94)}
}
function loop(){if(!over)update();draw();requestAnimationFrame(loop)}
let dragging=false;
cv.addEventListener('pointerdown',e=>{if(over||arrow||arrowsLeft<=0)return;dragging=true;pull={x:e.clientX,y:e.clientY}});
cv.addEventListener('pointermove',e=>{if(dragging)pull={x:e.clientX,y:e.clientY}});
cv.addEventListener('pointerup',e=>{if(!dragging)return;dragging=false;
  if(pull)shoot(e.clientX-bow.x,e.clientY-bow.y)});
cv.addEventListener('pointercancel',()=>{dragging=false;pull=null});
$('again').onclick=setup;
setup();
loop();
if('serviceWorker' in navigator)navigator.serviceWorker.register('sw.js');
