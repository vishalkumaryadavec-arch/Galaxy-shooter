function fits(shape,br,bc){
  for(let r=0;r<shape.length;r++)for(let c=0;c<shape[0].length;c++){
    if(!shape[r][c])continue;
    const rr=br+r,cc=bc+c;
    if(rr<0||cc<0||rr>=N||cc>=N||grid[rr][cc])return false;
  }
  return true;
}
function fitsAnywhere(shape){
  for(let r=0;r<N;r++)for(let c=0;c<N;c++)if(fits(shape,r,c))return true;
  return false;
}

  tray[pi]=null;

  // line clears
