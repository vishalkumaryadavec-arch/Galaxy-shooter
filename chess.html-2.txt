  if(!ms.length)return attacked(kingSq(w),!w)?-100000+(3-depth):0;
  let best=-Infinity;
  for(const m of orderMoves(ms)){
    const sv=B.slice(),ec={...castle},eep=ep;
    applyMove(m,'Q');
    const v=-search(depth-1,-beta,-alpha,!w);
    B=sv;castle=ec;ep=eep;
    if(v>best)best=v;if(best>alpha)alpha=best;if(alpha>=beta)break;
  }
  return best;
}
