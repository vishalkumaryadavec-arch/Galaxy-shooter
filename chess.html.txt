  if(m.from===7||m.to===7)castle.bk=false;if(m.from===0||m.to===0)castle.bq=false;
}
function legalMoves(from){
  const p=B[from];if(!p)return[];
  const w=isW(p);
  return pseudo(from).filter(m=>{
    const save=B.slice(),ec={...castle},eep=ep;
    applyMove(m,'Q');
    const ok=!attacked(kingSq(w),!w);
    B=save;castle=ec;ep=eep;
    return ok;
  });
}
function allLegal(w){
  const out=[];
  for(let i=0;i<64;i++)if(B[i]&&isW(B[i])===w)out.push(...legalMoves(i));
  return out;
}
