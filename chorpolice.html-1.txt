function blocked(x,y,r){
  const c0=Math.floor(x-r),c1=Math.floor(x+r),r0=Math.floor(y-r),r1=Math.floor(y+r);
  for(let cy=r0;cy<=r1;cy++)for(let cx=c0;cx<=c1;cx++)if(isWall(cx,cy)){

    if((x-nx)*(x-nx)+(y-ny)*(y-ny)<r*r)return true}
  return false;
}

    if(p.life<=0)parts.splice(i,1)}
  fireflies.forEach(f=>{f.ph+=dt;f.x+=Math.sin(f.ph*.7)*f.sp*dt;f.y+=Math.cos(f.ph*.5)*f.sp*dt;
    if(f.x<0)f.x=COLS;if(f.x>COLS)f.x=0;if(f.y<0)f.y=ROWS;if(f.y>ROWS)f.y=0});
}
function step(dt){
  cricketT-=dt;if(cricketT<=0){cricket();cricketT=1.8+Math.random()*3.5}
  const spd=2.7;
  if(joy&&(joy.dx||joy.dy)){
    const nx=px+joy.dx*spd*dt,ny=py+joy.dy*spd*dt;
    if(!blocked(nx,py,.3))px=nx;
    if(!blocked(px,ny,.3))py=ny;
    stepT-=dt;if(stepT<=0){nBurst(.05,.09,1100);beep(260+Math.random()*60,.03,'square',.04);stepT=.26}
  }
  // camera follows chor
  const tY=Math.max(0,Math.min(SH-H,py*S-H/2));camY+=(tY-camY)*Math.min(1,dt*8);
  if(exitC&&Math.floor(px)===exitC[0]&&Math.floor(py)===exitC[1])return levelClear();
  const hidden=inCart(px,py);
  let maxAlert=0;
  guards.forEach(g=>{
    if(g.alert>.55){
      const dx=px-g.x,dy=py-g.y,d=Math.hypot(dx,dy)||1;
      g.dir=Math.atan2(dy,dx);
      const nx=g.x+dx/d*g.spd*1.5*dt,ny=g.y+dy/d*g.spd*1.5*dt;
      if(!blocked(nx,g.y,.3))g.x=nx;if(!blocked(g.x,ny,.3))g.y=ny;
    }else{
      if(g.pause>0){g.pause-=dt;g.dir+=dt*1.2}
      else{
        const t=g.path[g.wi%g.path.length],tx=t[0]+.5,ty=t[1]+.5,dx=tx-g.x,dy=ty-g.y,d=Math.hypot(dx,dy);
        if(d<.08){g.wi=(g.wi+1)%g.path.length;g.pause=.8}
        else{g.dir=Math.atan2(dy,dx);g.x+=dx/d*g.spd*dt;g.y+=dy/d*g.spd*dt}
      }
    }
    const dx=px-g.x,dy=py-g.y,d=Math.hypot(dx,dy);
    let sees=false;
    if(d<g.range){let da=Math.atan2(dy,dx)-g.dir;while(da>Math.PI)da-=2*Math.PI;while(da<-Math.PI)da+=2*Math.PI;
      if(Math.abs(da)<FOV/2&&los(g.x,g.y,px,py)&&!(hidden&&d>0.9))sees=true}
    if(d<.55)sees=true;
    if(sees){g.alert=Math.min(1,g.alert+dt*1.7)}else g.alert=Math.max(0,g.alert-dt*.55);
    maxAlert=Math.max(maxAlert,g.alert);
    if(g.alert>=1){killer=g;gameOver('Police ki torch mein aa gaye, goli chal gayi!')}
  });
  if(maxAlert>.3&&prevAlert<=.3){beep(880,.09,'square',.13);setTimeout(()=>beep(622,.12,'square',.11),80)}
  prevAlert=maxAlert;
  if(maxAlert>.3){alertSnd-=dt;if(alertSnd<=0){beep(600+maxAlert*500,.06,'square',.07);alertSnd=.34-maxAlert*.2}
    $('status').textContent='⚠️ Police dekh rahi hai - BHAGO ya CHHUPO!'}
  else if(!msg)$('status').textContent=hidden?'🛺 Thele ke peeche chhupe ho - safe!':'Gali mein chhup ke jao, gate tak pahuncho!';
  if(msg){msgT+=dt;if(msgT>1.6){msg='';msgT=0}}
}
