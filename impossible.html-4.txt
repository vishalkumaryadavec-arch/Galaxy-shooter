function spawn(dt){
  const rate=8+Math.min(t,50)*.4+(t>50?(t-50)*.15:0); // bullets/sec ramp
  spawnAcc+=dt*rate;
  while(spawnAcc>=1){
    spawnAcc--;
    const sp=3.2+t*.09; // speed ramp
    const ph=Math.floor(t/12)%7;
    if(ph===0||ph===4||ph===6){ // spiral from top center
      ang+=.55;
      const dx=Math.cos(ang)*.55,dy=Math.abs(Math.sin(ang))*.6+.75;
      const m=Math.hypot(dx,dy);
      bullets.push({x:W/2,y:-8,vx:dx/m*sp,vy:dy/m*sp,c:'#f0f'});
    }
    if(ph===1||ph===4||ph===5||ph===6){ // seeded rain
      bullets.push({x:rnd()*W,y:-8,vx:(rnd()-.5)*1.2,vy:sp*1.15,c:'#0ff'});
    }
    if(ph===2||ph===5||ph===6){ // side waves
      const fromL=rnd()<.5;
      bullets.push({x:fromL?-8:W+8,y:H*(.15+rnd()*.5),vx:(fromL?1:-1)*sp*.95,vy:sp*.35,c:'#ffd60a'});
    }
    if((ph===3||ph===5||ph===6)&&ship){ // aimed shots from top corners
      const sx=rnd()<.5?0:W,dx=ship.x-sx,dy=ship.y-(-8),m=Math.hypot(dx,dy);
      bullets.push({x:sx,y:-8,vx:dx/m*sp*1.05,vy:dy/m*sp*1.05,c:'#f55'});
    }
  }
}
