const $=id=>document.getElementById(id);
const TRACK=[[6,1],[6,2],[6,3],[6,4],[6,5],[5,6],[4,6],[3,6],[2,6],[1,6],[0,6],[0,7],[0,8],[1,8],[2,8],[3,8],[4,8],[5,8],[6,9],[6,10],[6,11],[6,12],[6,13],[6,14],[7,14],[8,14],[8,13],[8,12],[8,11],[8,10],[8,9],[9,8],[10,8],[11,8],[12,8],[13,8],[14,8],[14,7],[14,6],[13,6],[12,6],[11,6],[10,6],[9,6],[8,5],[8,4],[8,3],[8,2],[8,1],[8,0],[7,0],[6,0]];

function buildBoard(){
  const b=$('board');b.innerHTML='';S=b.clientWidth/15;
  for(let r=0;r<15;r++)for(let c=0;c<15;c++){
    const d=document.createElement('div');d.className='cell';
    d.style.cssText='left:'+(c*S)+'px;top:'+(r*S)+'px;width:'+S+'px;height:'+S+'px';
    let bg=null;
    for(const col in BASEO){const[oR,oC]=BASEO[col];
      if(r>=oR&&r<oR+6&&c>=oC&&c<oC+6){
        if(r>=oR+1&&r<oR+5&&c>=oC+1&&c<oC+5)bg='#141a38';
        else bg=COL[col];
        SPOTS.forEach(([sr,sc],k)=>{if(r===oR+sr&&c===oC+sc){bg='#141a38';d.style.borderRadius='50%';d.style.border='2px solid '+COL[col]}});
      }}
    if(r>=6&&r<=8&&c>=6&&c<=8){bg='linear-gradient(135deg,#ff3b5c,#2ee66b,#ffd60a,#3ba7ff)'}
    TRACK.forEach(([tr,tc],i)=>{if(tr===r&&tc===c){
      bg='#e8ecff';
      if(SAFE.has(i))d.classList.add('star');
      if(i===START.red)bg=COL.red;if(i===START.green)bg=COL.green;if(i===START.yellow)bg=COL.yellow;if(i===START.blue)bg=COL.blue;
    }});
    for(const col in HOMEP)HOMEP[col].forEach(([hr,hc])=>{if(hr===r&&hc===c)bg=COL[col]});
    if(bg)d.style.background=bg;
    b.appendChild(d);
  }
}
function cellOf(p,pos,ti){
  if(pos===-1){const[oR,oC]=BASEO[p.color];return[oR+SPOTS[ti][0],oC+SPOTS[ti][1]]}
  if(pos<=51)return TRACK[(p.start+pos)%52];
  if(pos<=56)return HOMEP[p.color][pos-52];
  return[7,7];
}
// ---------- rules ----------
function tokensOnAbs(abs){const r=[];P.forEach((p,pi)=>p.tokens.forEach((pos,ti)=>{if(pos>=0&&pos<=51&&(p.start+pos)%52===abs)r.push({pi,ti,color:p.color})}));return r}
function blockAt(abs,forColor){const t=tokensOnAbs(abs),byC={};t.forEach(x=>byC[x.color]=(byC[x.color]||0)+1);for(const c in byC)if(byC[c]>=2&&c!==forColor)return true;return false}
function canMove(pi,ti,d){
  const p=P[pi],pos=p.tokens[ti];
  if(pos===57)return false;
  if(pos===-1)return d===6&&!blockAt(p.start,p.color);
  const np=pos+d;if(np>57)return false;
  const lim=Math.min(np,51);
  for(let s=pos+1;s<=lim;s++)if(blockAt((p.start+s)%52,p.color))return false;
  if(np<=51&&blockAt((p.start+np)%52,p.color))return false;
  return true;
}
function legalMoves(pi,d){const r=[];for(let ti=0;ti<4;ti++)if(canMove(pi,ti,d))r.push(ti);return r}
function applyMove(pi,ti,d){
  const p=P[pi];let pos=p.tokens[ti];let captured=0,home=false;
  pos=pos===-1?0:pos+d;p.tokens[ti]=pos;
  if(pos<=51){const abs=(p.start+pos)%52;
    if(!SAFE.has(abs))tokensOnAbs(abs).forEach(x=>{if(x.pi!==pi){P[x.pi].tokens[x.ti]=-1;captured++}});
  }
  if(pos===57)home=true;
  return{captured,home};
}
// ---------- AI ----------
function aiPick(pi,d){
  const p=P[pi],cands=legalMoves(pi,d);let best=cands[0],bs=-1e9;
  for(const ti of cands){
    const pos=p.tokens[ti];let sc=Math.random()*6;
    if(pos===-1)sc+=30;
    const np=pos===-1?0:pos+d;
    if(np===57)sc+=40;
    if(np<=51){const abs=(p.start+np)%52;
      if(!SAFE.has(abs))sc+=55*tokensOnAbs(abs).filter(x=>x.pi!==pi).length;
      if(SAFE.has(abs))sc+=25;
      if(pos>=0&&pos<=51&&SAFE.has((p.start+pos)%52))sc-=12;
      // danger: enemy behind within 6 on unsafe
      if(!SAFE.has(abs))for(let back=1;back<=6;back++){const behind=(abs-back+52)%52;
        if(tokensOnAbs(behind).some(x=>x.pi!==pi))sc-=18}
    }
    sc+=np*.15;
    if(sc>bs){bs=sc;best=ti}
  }
  return best;
}

  const d=$('dice');d.classList.toggle('rolling',!!rolling);
  d.innerHTML='';for(let i=0;i<9;i++){const p=document.createElement('div');p.className='pip'+(PIPS[n]&&PIPS[n].includes(i)?' on':'');d.appendChild(p)}
}

  document.querySelectorAll('.tok').forEach(t=>t.remove());
  const b=$('board');const groups={};
  P.forEach((p,pi)=>p.tokens.forEach((pos,ti)=>{
    const[r,c]=cellOf(p,pos,ti);const k=r+','+c;(groups[k]=groups[k]||[]).push({pi,ti});
  }));
  for(const k in groups){

    const n=g.length;let sz=n===1?.72:n<=4?.44:.3;
    g.forEach(({pi,ti},idx)=>{
      const t=document.createElement('div');t.className='tok'+(pi===0?' me':'')+(phase==='move'&&pi===0&&movables.includes(ti)?' mv':'');

      t.onclick=()=>{if(phase==='move'&&turn===0&&movables.includes(ti)&&pi===0)doUserMove(ti)};
      b.appendChild(t);
    });
  }
}
