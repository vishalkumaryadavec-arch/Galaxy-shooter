  if(d===6){sixes++;if(sixes>=3){status(NAME[P[pi].color]+': teen 6 lagataar! Baari gayi 😅');beep(150,.35,'sawtooth',.09);return setTimeout(nextTurn,1200)}}

}
function doMove(pi,ti){
  phase='anim';movables=[];render();
  const p=P[pi],from=p.tokens[ti];
  const steps=from===-1?1:dice;
  let i=0;
  const stepFn=()=>{
    p.tokens[ti]=from===-1?0:from+i+1;
    tick(i);if(navigator.vibrate)navigator.vibrate(12);
    render();i++;
    if(i<steps)setTimeout(stepFn,190);
    else setTimeout(()=>finishMove(pi,ti),200);
  };
  stepFn();
}
function finishMove(pi,ti){
  const{captured,home}=landingEvents(pi,ti);render();
  let extra=dice===6;
  if(captured){extra=true;earn(5*captured);beep(200,.2,'sawtooth',.1);setTimeout(()=>beep(140,.25,'sawtooth',.09),120);setTimeout(()=>beep(90,.3,'sawtooth',.07),240);if(navigator.vibrate)navigator.vibrate([60,40,60]);status('⚔️ '+NAME[P[pi].color]+' ne '+(captured>1?captured+' gotiyan':'goti')+' kaati! +'+5*captured+' 🪙 Extra chance!')}
  else if(home){extra=true;earn(2);beep(1000,.18,'sine',.1);setTimeout(()=>beep(1300,.22,'sine',.09),140);setTimeout(()=>beep(1600,.25,'sine',.08),280);status('🏠 '+NAME[P[pi].color]+' ki goti ghar! +2 🪙 Extra chance!')}
  if(P[pi].tokens.every(x=>x===57))return setTimeout(()=>gameOver(pi),700);

  },captured||home?1100:350);
}
