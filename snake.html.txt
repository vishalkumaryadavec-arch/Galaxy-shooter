function steer(dx,dy){
  if(Math.abs(dx)<12&&Math.abs(dy)<12)return;

}

function step(){
  if(pend.length)dir=pend.shift();
  const h={x:snake[0].x+dir.x,y:snake[0].y+dir.y};
  if(h.x<0||h.y<0||h.x>=COLS||h.y>=ROWS||snake.some(s=>s.x===h.x&&s.y===h.y)){endGame();return}
  snake.unshift(h);
  if(food&&h.x===food.x&&h.y===food.y){
    eaten++;flash=1;eatS();
    $('tEl').textContent=eaten;
    $('pfill').style.width=(eaten/GOAL*100)+'%';
    if(eaten>=GOAL){winGame();return}
    nextFood();
  }else snake.pop();
}

  const dt=Math.min((now-lastTs)/1000,.05);lastTs=now;
  stars.forEach(s=>{s.y+=s.s*dt*20;if(s.y>H)s.y=0});
  if(flash>0)flash-=dt*3;
  if(!running)return;
  acc+=dt*1000;
  const ms=stepMs();
  while(acc>=ms&&running){acc-=ms;step()}
}
