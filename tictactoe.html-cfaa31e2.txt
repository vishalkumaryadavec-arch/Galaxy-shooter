<script>
const $=id=>document.getElementById(id);
let board=Array(9).fill(''),over=false,hard=true;
let wins=+localStorage.getItem('ttt_wins')||0,loss=+localStorage.getItem('ttt_losses')||0,draws=+localStorage.getItem('ttt_draws')||0;
const LINES=[[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];
let AC;function beep(f,d,t='square',v=.05){try{AC=AC||new(window.AudioContext||window.webkitAudioContext)();const o=AC.createOscillator(),g=AC.createGain();o.type=t;o.frequency.value=f;g.gain.value=v;o.connect(g);g.connect(AC.destination);o.start();g.gain.exponentialRampToValueAtTime(.0001,AC.currentTime+d);o.stop(AC.currentTime+d)}catch(e){}}
function showScores(){$('sw').textContent=wins;$('sd').textContent=draws;$('sl').textContent=loss}
function build(){const b=$('board');b.innerHTML='';for(let i=0;i<9;i++){const c=document.createElement('div');c.className='cell';c.onclick=()=>move(i);b.appendChild(c)}}
function paint(){document.querySelectorAll('.cell').forEach((c,i)=>{c.textContent=board[i]==='X'?'❌':board[i]==='O'?'⭕':'';c.classList.toggle('x',board[i]==='X');c.classList.toggle('o',board[i]==='O')})}
function winner(b){for(const L of LINES){if(b[L[0]]&&b[L[0]]===b[L[1]]&&b[L[1]]===b[L[2]])return{p:b[L[0]],line:L}}return b.includes('')?null:{p:'D'}}
function minimax(b,isMax,depth){
  const w=winner(b);if(w){if(w.p==='O')return 10-depth;if(w.p==='X')return depth-10;return 0}
  let best=isMax?-99:99;
  for(let i=0;i<9;i++){if(!b[i]){b[i]=isMax?'O':'X';const s=minimax(b,!isMax,depth+1);b[i]='';best=isMax?Math.max(best,s):Math.min(best,s)}}
  return best;
}
function aiMove(){
  if(over)return;$('status').textContent='AI soch raha hai... 🤖';
  setTimeout(()=>{
    let pick;
    const empty=board.map((v,i)=>v?null:i).filter(v=>v!==null);
    if(!hard&&Math.random()<.7){pick=empty[Math.floor(Math.random()*empty.length)]}
    else{let bs=-99;const opts=[];for(const i of empty){board[i]='O';const s=minimax(board.slice?board:board, false, 0);board[i]='';if(s>bs){bs=s;opts.length=0;opts.push(i)}else if(s===bs)opts.push(i)}pick=opts[Math.floor(Math.random()*opts.length)]}
    board[pick]='O';beep(300,.1,'square',.05);paint();finish()||($('status').textContent='Tumhari baari (❌)');
  },350);
}
function move(i){
  if(over||board[i])return;
  board[i]='X';beep(600,.08,'square',.05);paint();
  if(!finish())aiMove();
}
function finish(){
  const w=winner(board);
  if(!w)return false;
  over=true;
  if(w.line)w.line.forEach(i=>document.querySelectorAll('.cell')[i].classList.add('win'));
  if(w.p==='X'){$('status').textContent='🎉 Tum jeet gaye!';wins++;localStorage.setItem('ttt_wins',wins);beep(880,.15,'sine',.07);setTimeout(()=>beep(1320,.25,'sine',.07),150)}
  else if(w.p==='O'){$('status').textContent='🤖 AI jeet gaya!';loss++;localStorage.setItem('ttt_losses',loss);beep(180,.3,'sawtooth',.07)}
  else{$('status').textContent='😐 Draw ho gaya';draws++;localStorage.setItem('ttt_draws',draws);beep(440,.2,'triangle',.06)}
  if(navigator.vibrate)navigator.vibrate(80);
  showScores();$('again').classList.remove('hide');
  return true;
}
function reset(){board=Array(9).fill('');over=false;$('status').textContent='Tumhari baari (❌)';$('again').classList.add('hide');build()}
$('again').onclick=reset;
$('d-easy').onclick=()=>{hard=false;$('d-easy').classList.add('on');$('d-hard').classList.remove('on');reset()};
$('d-hard').onclick=()=>{hard=true;$('d-hard').classList.add('on');$('d-easy').classList.remove('on');reset()};
build();showScores();
if('serviceWorker' in navigator)navigator.serviceWorker.register('sw.js');
