function genLevel(lv){
  const n=Math.min(4+Math.floor((lv-1)/2),9); // L1-2:4 colors, phir har 2 level pe +1, max 9

    tubes[to].push(tubes[from].pop());
    last=from+'-'+to;

  // if accidentally already solved, scramble more
  if(isWin())genLevel(lv);
}
