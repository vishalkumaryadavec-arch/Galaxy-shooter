function genLevel(lv){
  const n=Math.min(3+lv,9); // colors: level1=4 ... level6+=9
  // start solved, scramble with random single-unit moves -> solvable
  tubes=[];
  for(let c=0;c<n;c++)tubes.push(Array(CAP).fill(c));
  tubes.push([]);tubes.push([]);
  let last='';
  const steps=60+lv*25;
  for(let s=0;s<steps;s++){
    const from=Math.floor(Math.random()*tubes.length);
    if(!tubes[from].length)continue;
    const to=Math.floor(Math.random()*tubes.length);
    if(to===from||tubes[to].length>=CAP)continue;
    if(last===to+'-'+from){continue}
    // move only ONE unit so mixing is fine-grained
    tubes[to].push(tubes[from].pop());
    last=from+'-'+to;
  }
  // if accidentally already solved, scramble more
  if(isWin())genLevel(lv);
}
function topRun(t){if(!t.length)return 0;const c=t[t.length-1];let n=0;for(let i=t.length-1;i>=0&&t[i]===c;i--)n++;return n}
function canPour(a,b){
  if(a===b||!tubes[a].length||tubes[b].length>=CAP)return 0;
  const c=tubes[a][tubes[a].length-1];
  if(tubes[b].length&&tubes[b][tubes[b].length-1]!==c)return 0;
  return Math.min(topRun(tubes[a]),CAP-tubes[b].length);
}
function pour(a,b){

  hist.push({a,b,n});

  moves++;beep(500+n*80,.12,'sine',.06);
  if(navigator.vibrate)navigator.vibrate(30);
  return true;
}
